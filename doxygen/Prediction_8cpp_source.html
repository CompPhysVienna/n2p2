50void Prediction::readStructureFromFile(string const& fileName)
51{
52 ifstream file;
53 file.open(fileName.c_str());
54 structure.reset();
55 structure.setElementMap(elementMap);
56 structure.readFromFile(fileName);
57 removeEnergyOffset(structure);
58 if (normalize)
59 {
60 structure.toNormalizedUnits(meanEnergy, convEnergy, convLength, convCharge);
61 }
62 file.close();
63
64 return;
65}

67void Prediction::predict()
68{
69 evaluateNNP(structure);
70 if (normalize)
71 {
72 structure.toPhysicalUnits(meanEnergy, convEnergy, convLength, convCharge);
73 }
74 addEnergyOffset(structure, false);
75 addEnergyOffset(structure, true);
76
77 return;
78}
